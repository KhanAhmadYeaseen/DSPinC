FFT(x, N):

    if N == 1:
        return x

    ─────────────────────────────
    1. SPLIT
    ─────────────────────────────

    even = [x[0], x[2], x[4], ..., x[N-2]]
    odd  = [x[1], x[3], x[5], ..., x[N-1]]


    ─────────────────────────────
    2. GO DEEPER INTO RECURSION
    ─────────────────────────────

	E[0 ... N/2-1] = FFT(even, N/2)
	O[0 ... N/2-1] = FFT(odd,  N/2)


    ─────────────────────────────
    3. BACKTRACK / COMBINE
    ─────────────────────────────

    for k = 0 to N/2 - 1:

        W = exp(-j 2πk/N)

        T = W × O[k]

        X[k]       = E[k] + T
        X[k + N/2] = E[k] - T


    return X
